// NONCOMPILING SPEC. Names, iteration and failure syntax are proposals only.
void collect(u8[] first_bytes, u8[] second_bytes) throws Error {
	var bag = new DiagnosticBag(2);             // Allocate capacity explicitly.
	var first = Utf8Message.from_utf8(first_bytes); // Validate and copy bytes.
	var second = Utf8Message.from_utf8(second_bytes); // Independent owner.
	bag.append(move Diagnostic(2, 100, 10, move first));
	bag.append(move Diagnostic(1, 200, 20, move second));
	for (borrowed Diagnostic entry in bag.inspect()) {
		render(entry.severity, entry.code, entry.offset, entry.message);
	}                                         // All inspection loans end here.
	try {
		bag.reserve(3);                         // Visible allocation/relocation.
	} catch (CapacityLimit error) {
		// Retained count/order/owners unchanged. Actual OOM policy is separate.
		render(error.message);
		return;
	}
	var third = Utf8Message.from_utf8(first_bytes);
	bag.append(move Diagnostic(2, 300, 30, move third)); // No growth at commit.
	// No view survives mutation. No implicit refcount, byte/text conversion,
	// normalization or claim of recovering from today's fatal allocator.
}                                             // Retire entries then storage.
void scenario() throws Error {
	u8[5] composed = {99, 97, 102, 195, 169};    // café: 5 bytes.
	u8[6] combining = {99, 97, 102, 101, 204, 129}; // cafe + combining acute.
	u8[4] symbol = {240, 159, 150, 164};         // 🖤: 4 bytes.
	collect(slice(composed), slice(composed));  // Equal content, two owners.
	collect(slice(combining), slice(symbol));   // No normalization requirement.
}
